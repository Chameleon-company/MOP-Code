Scenario

In a smart city context, both commuters and transportation authorities require accurate and timely insights into traffic conditions. Rather than relying solely on past traffic trends, this system leverages both temporal patterns and environmental conditions to provide more intelligent predictions of congestion.

For instance, congestion levels often follow clear daily cycles, with higher traffic during morning and evening peak hours and lower activity during midday. However, these patterns can be influenced by environmental factors such as temperature, humidity, or wind conditions. By integrating these variables, the system can better capture variations in traffic behavior across different locations and times.

In practice, the system can predict whether a specific area is likely to be congested at a given time and support recommendations such as optimal travel times or alternative routes. Additionally, it provides valuable insights for city planners to understand how environmental conditions interact with urban mobility, ultimately contributing to more efficient and sustainable transportation systems.

Datasets Used

1. Transport Activity Counts Dataset Link: https://data.melbourne.vic.gov.au/explore/dataset/transport-activity-counts/information/

This dataset contains traffic activity counts in 5-minute intervals from AIRS (Artificial Intelligence Road Survey) sensor devices located across the city. AIRS is an AI-based service that autonomously detects and classifies road users and their interaction with the road environment. The data is updated hourly and can be used to analyse variations in transport activity across the day.

The dataset includes:

  • Traffic counts from multiple monitoring locations
  • Different transportation classes such as pedestrians, cyclists, cars, taxis, buses, vans, and trucks
  • Geographic coordinates of monitoring sensors
  • High-frequency temporal traffic observations

2. Microclimate Sensors Data Dataset Link: https://data.melbourne.vic.gov.au/explore/dataset/microclimate-sensors-data/information/

This dataset contains climate readings from climate sensors located within the City. The data is updated every fifteen minutes and can be used to determine variations in microclimate changes throughout the day.

The dataset includes:

  • Air temperature
  • Relative humidity
  • Atmospheric pressure
  • Wind speed and wind direction
  • PM2.5 and PM10 pollution measurements
  • Environmental noise levels

These environmental variables provide valuable contextual information that may influence transportation patterns and congestion behavior.

def cap_outliers(df, cols):
    for col in cols:
        q1 = df[col].quantile(0.25)
        q3 = df[col].quantile(0.75)
        iqr = q3 - q1
        lower = q1 - 1.5 * iqr
        upper = q3 + 1.5 * iqr
        df[col] = df[col].clip(lower, upper)
    return df

# Normalize
def normalize_text(s: str) -> str:
    s = str(s).lower()
    s = re.sub(r"[^a-z0-9\s\.\-]", " ", s)
    s = re.sub(r"\s+", " ", s).strip()
    return s


# Extract scalar safely from DataFrame
def get_scalar(df, col, default=None):
    if col not in df.columns:
        return default
    return df[col].iloc[0]


# Location matching
def location_match(input_loc, response_text, threshold=80):
    if input_loc is None:
        return False, 0.0

    input_norm = normalize_text(input_loc)
    response_norm = normalize_text(response_text)

    if input_norm and input_norm in response_norm:
        return True, 100.0

    score = fuzz.partial_ratio(input_norm, response_norm)
    return score >= threshold, float(score)


# Numeric value matching with tolerance
def numeric_value_match(value, response_text, tolerance=0.02):
    text = response_text.lower()

    try:
        value = float(value)
    except (TypeError, ValueError):
        return False, 0.0

    candidates = re.findall(r"-?\d+(?:\.\d+)?", text)
    candidates = [float(x) for x in candidates]

    for c in candidates:
        if abs(c - value) <= max(tolerance, abs(value) * tolerance):
            return True, 1.0

    return False, 0.0


# Feature coverage
def feature_coverage(df_input, response_text, raw_input=None):
    row = df_input.iloc[0]
    coverage = {}

    for col in df_input.columns:
        value = row[col]

        if pd.api.types.is_numeric_dtype(df_input[col]):
            mentioned, score = numeric_value_match(value, response_text)
        else:
            mentioned = normalize_text(value) in normalize_text(response_text)
            score = 1.0 if mentioned else 0.0

        coverage[col] = {
            "value": value,
            "mentioned": bool(mentioned),
            "score": float(score)
        }

    # Optional explanation-only fields
    if raw_input is not None:
        optional_fields = ["hour", "day_of_week", "is_weekend", "countLocationName"]

        for col in optional_fields:
            if col not in raw_input:
                continue

            value = raw_input[col]

            if col == "countLocationName":
                mentioned, loc_score = location_match(value, response_text)
                score = loc_score / 100
            elif isinstance(value, (int, float, np.integer, np.floating)):
                mentioned, score = numeric_value_match(value, response_text)
            else:
                mentioned = normalize_text(value) in normalize_text(response_text)
                score = 1.0 if mentioned else 0.0

            coverage[col] = {
                "value": value,
                "mentioned": bool(mentioned),
                "score": float(score)
            }

    coverage_score = np.mean([c["score"] for c in coverage.values()])

    return coverage, float(coverage_score)


# Traffic classification check
def check_traffic_classification(response_text: str):
    text = normalize_text(response_text)

    label_keywords = {
        "low": ["low", "normal", "not congested", "non congested"],
        "medium": ["medium", "moderate"],
        "high": ["high", "congested", "heavy", "severe"]
    }

    found_labels = []

    for label, keywords in label_keywords.items():
        if any(normalize_text(keyword) in text for keyword in keywords):
            found_labels.append(label)

    return {
        "found_labels": found_labels,
        "valid": len(found_labels) > 0
    }


# Prediction consistency check
def check_prediction_consistency(response_text, prediction=None):
    if prediction is None:
        return {
            "valid": True,
            "score": 1.0,
            "reason": "No prediction provided for consistency check."
        }

    text = normalize_text(response_text)

    congested_terms = ["congested", "high", "heavy", "severe"]
    normal_terms = ["low", "normal", "not congested", "non congested"]

    has_congested = any(normalize_text(t) in text for t in congested_terms)
    has_normal = any(normalize_text(t) in text for t in normal_terms)

    if prediction == 1:
        valid = has_congested
    else:
        valid = has_normal

    return {
        "valid": bool(valid),
        "score": 1.0 if valid else 0.0,
        "reason": "Response matches model prediction." if valid else "Response does not clearly match model prediction."
    }


# Main evaluator
def evaluate_response(
    df_input,
    response_text,
    raw_input=None,
    prediction=None,
    location_col="countLocationName"
):
    feature_cov, feature_score = feature_coverage(
        df_input=df_input,
        response_text=response_text,
        raw_input=raw_input
    )

    location_value = None
    if raw_input is not None and location_col in raw_input:
        location_value = raw_input[location_col]
    elif location_col in df_input.columns:
        location_value = get_scalar(df_input, location_col)

    location_ok, loc_score = location_match(location_value, response_text)

    traffic_cls = check_traffic_classification(response_text)
    prediction_check = check_prediction_consistency(response_text, prediction)

    overall_score = (
        feature_score * 0.45 +
        (loc_score / 100) * 0.25 +
        (1.0 if traffic_cls["valid"] else 0.0) * 0.15 +
        prediction_check["score"] * 0.15
    )

    result = {
        "feature_coverage": feature_cov,
        "feature_score": float(feature_score),
        "location_match": {
            "ok": bool(location_ok),
            "score": float(loc_score)
        },
        "traffic_classification_valid": traffic_cls["valid"],
        "traffic_labels_found": traffic_cls["found_labels"],
        "prediction_consistency": prediction_check,
        "overall_score": float(overall_score)
    }

    return result

rag_texts = [doc["title"] + ". " + doc["text"] for doc in rag_documents]

vectorizer = TfidfVectorizer()
rag_matrix = vectorizer.fit_transform(rag_texts)

def retrieve_rag_context(query, top_k=5):
    query_vec = vectorizer.transform([query])
    similarities = cosine_similarity(query_vec, rag_matrix).flatten()
    top_indices = similarities.argsort()[::-1][:top_k]

    retrieved_docs = []
    for idx in top_indices:
        retrieved_docs.append(rag_documents[idx])

    context = "\n".join(
        [f"- {doc['title']}: {doc['text']}" for doc in retrieved_docs]
    )

    return context
