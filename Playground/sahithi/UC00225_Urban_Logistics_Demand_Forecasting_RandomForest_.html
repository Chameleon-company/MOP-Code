all_dfs = []

for url in urls:
    r = requests.get(url)
    z = zipfile.ZipFile(io.BytesIO(r.content))

    for file in z.namelist():
        if file.endswith(".csv"):
            try:
                df = pd.read_csv(z.open(file), encoding="utf-8")
            except UnicodeDecodeError:
                df = pd.read_csv(z.open(file), encoding="latin1")
            all_dfs.append(df)

final_df = pd.concat(all_dfs, ignore_index=True)
final_df = final_df.loc[:, ~final_df.columns.str.contains("^Unnamed")]

Encoding Categorical Features: The day_of_week column contained text values, which cannot be directly used in machine learning models. To solve this, label encoding was applied to convert the days into numerical values. An additional is_weekend feature was also created to help the model identify differences between weekday and weekend logistics demand patterns.

In [ ]:
from sklearn.preprocessing import LabelEncoder

# Encode day_of_week
le = LabelEncoder()

van_truck_df["day_of_week_encoded"] = le.fit_transform(
    van_truck_df["day_of_week"]
)

# Weekend feature
van_truck_df["is_weekend"] = van_truck_df["day_of_week"].isin(
    ["Saturday", "Sunday"]
).astype(int)

Cross Validation: K-Fold Cross Validation was applied to evaluate the Random Forest model on multiple data splits. A sample dataset was used to reduce computation time while still testing model consistency and reliability. This approach provides a more robust estimate of model performance compared to using only a single train-test split.

from sklearn.model_selection import cross_val_score
import numpy as np

# Take sample data for faster cross-validation
sample_df = van_truck_df.sample(50000, random_state=42)

# Cross-validation
cv_scores = cross_val_score(
    rf_model,
    X_sample,
    y_sample,
    cv=5,
    scoring="neg_mean_absolute_error",
    n_jobs=-1
)

# Convert negative scores to positive
cv_mae = -cv_scores
